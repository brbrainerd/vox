# Test-Suite Performance & Gate Tier Redesign — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Cut local `vox ci pre-push --full --with-coverage` from ~590s to ≤260s by (1) speeding up `vox-arch-check` from 165s to ≤15s with a persistent cache, (2) moving three slow workspace tests to an opt-in partition, (3) extending `vox ci pre-push` with `--with-coverage`, `--since`, and `--include-slow` flags, and (4) shipping a reusable Vox measurement harness.

**Architecture:** Two cooperating Rust subsystems — `vox-arch-check` gains an on-disk arch-graph cache keyed on `Cargo.lock` + `layers.toml` + `where-things-live.md` hashes, and `vox ci pre-push` gains three new flags wired into the existing step builder. A `scripts/perf/test-baseline.vox` harness measures every tier wall-clock and writes structured JSON + Markdown, providing a budget contract in `contracts/budgets/test-tier-budgets.v1.yaml`.

**Tech Stack:** Rust (`anyhow`, `serde`, `cargo-metadata`, existing `vox-cli` + `vox-arch-check` crates), Vox scripting (`process`, `fs`, `json` builtins), cargo-nextest 0.9.95, cargo-llvm-cov 0.8.5, PowerShell (Windows measurement), TOML/YAML.

**Baseline:** `docs/superpowers/specs/2026-05-27-test-suite-baseline-measurements.md`

### Task P0-2: Measurement harness Vox script

**Files:**
- Create: `scripts/perf/test-baseline.vox`

The harness times each gate via PowerShell `Measure-Command`, serialises results to JSON, and prints a Markdown summary.

- [ ] **Step 1: Create `scripts/perf/test-baseline.vox`**

```vox
// vox:caps fs process env
// scripts/perf/test-baseline.vox — measures wall-clock for every CI gate tier.
//
// Usage:
//   vox run scripts/perf/test-baseline.vox
//   vox run scripts/perf/test-baseline.vox -- --write-budgets
//
// Output:
//   contracts/reports/test-baseline-<YYYY-MM-DD>.json
//   (with --write-budgets) contracts/budgets/test-tier-budgets.v1.yaml
import fs;
import process;
import json;

fn measure_ms(cmd: str, args: list[str]) to int {
    // Use PowerShell Measure-Command for precise wall-clock timing on Windows.
    // Builds: powershell -NoProfile -Command "(Measure-Command { <cmd> <args> }).TotalMilliseconds"
    let inner = cmd + " " + args.join(" ")
    let ps_cmd = "(Measure-Command { " + inner + " 2>&1 | Out-Null }).TotalMilliseconds"
    let res = process.run("powershell", ["-NoProfile", "-Command", ps_cmd])
    if res is null { return -1 }
    let r = res.unwrap()
    if r.code != 0 { return -1 }
    let ms_str = r.stdout.trim()
    // PowerShell returns e.g. "12345.678" — truncate to int
    let dot_pos = ms_str.find(".")
    if dot_pos >= 0 {
        return int(ms_str.slice(0, dot_pos))
    }
    return int(ms_str)
}

fn run_and_record(label: str, cmd: str, args: list[str]) to Json {
    print("  timing: " + label + " ...")
    let ms = measure_ms(cmd, args)
    let ok = ms >= 0
    print("    -> " + str(ms) + "ms ok=" + str(ok))
    // Build a JSON object for this step
    let obj_str = "{\"label\":\"" + label + "\",\"wall_ms\":" + str(ms) + ",\"ok\":" + str(ok) + "}"
    return json.parse(obj_str).unwrap()
}

fn env_str(cmd: str, args: list[str]) to str {
    let res = process.run(cmd, args)
    if res is null { return "unknown" }
    let r = res.unwrap()
    if r.code != 0 { return "unknown" }
    return r.stdout.trim()
}

fn target_dir_gb() to float {
    let res = process.run("powershell", ["-NoProfile", "-Command",
        "$s=(Get-ChildItem target -Recurse -ErrorAction SilentlyContinue | Measure-Object -Property Length -Sum).Sum; [math]::Round($s/1GB,2)"])
    if res is null { return 0.0 }
    let r = res.unwrap()
    if r.code != 0 { return 0.0 }
    let s = r.stdout.trim()
    if s == "" { return 0.0 }
    return float(s)
}

fn make_tier(steps: list[Json]) to Json {
    let mut total = 0
    for s in steps {
        let ms = s.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
        if ms > 0 { total = total + ms }
    }
    // Manually construct JSON string since Vox has no object literal builder
    let mut steps_json = "["
    let mut first = true
    for s in steps {
        if not first { steps_json = steps_json + "," }
        let label = s.get("label").and_then(fn(j: Json) to Option[str] { j.as_str() }).unwrap_or("")
        let ms = s.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
        let ok = s.get("ok").and_then(fn(j: Json) to Option[bool] { j.as_bool() }).unwrap_or(false)
        steps_json = steps_json + "{\"label\":\"" + label + "\",\"wall_ms\":" + str(ms) + ",\"ok\":" + str(ok) + "}"
        first = false
    }
    steps_json = steps_json + "]"
    let tier_str = "{\"wall_ms\":" + str(total) + ",\"steps\":" + steps_json + "}"
    return json.parse(tier_str).unwrap()
}

fn main() {
    print("=== test-baseline.vox ===")

    let write_budgets = false
    // (Argument parsing: check env or extend when Vox gains argv support)

    // ── env snapshot ──────────────────────────────────────────────────────
    print("Collecting environment...")
    let cpu_threads = env_str("powershell", ["-NoProfile", "-Command",
        "(Get-CimInstance Win32_Processor).NumberOfLogicalProcessors"])
    let ram_gb = env_str("powershell", ["-NoProfile", "-Command",
        "[math]::Round((Get-CimInstance Win32_OperatingSystem).TotalVisibleMemorySize/1MB,2)"])
    let target_gb = str(target_dir_gb())
    let cargo_ver = env_str("cargo", ["--version"])
    let nextest_ver = env_str("cargo", ["nextest", "--version"])
    let llvmcov_ver = env_str("cargo", ["llvm-cov", "--version"])

    // ── per-gate measurements (serial to avoid target/ lock contention) ───

    // fast tier steps
    print("\n[fast tier]")
    let s_fmt = run_and_record("cargo fmt --check", "cargo", ["fmt", "--all", "--", "--check"])
    let s_ssot = run_and_record("vox-cli ci ssot-drift", "cargo", ["run", "-p", "vox-cli", "--quiet", "--", "ci", "ssot-drift"])
    let s_toestub = run_and_record("vox-cli ci toestub-scoped", "cargo", ["run", "-p", "vox-cli", "--quiet", "--", "ci", "toestub-scoped", "--mode", "enforce-warn"])
    let s_arch = run_and_record("scripts/arch-check.vox", "cargo", ["run", "-p", "vox-cli", "--quiet", "--", "run", "scripts/arch-check.vox"])
    let fast_steps = [s_fmt, s_ssot, s_toestub, s_arch]
    let tier_fast = make_tier(fast_steps)

    // complete tier adds clippy
    print("\n[complete tier (adds clippy)]")
    let s_clippy = run_and_record("cargo clippy --workspace --all-targets", "cargo", ["clippy", "--workspace", "--all-targets", "--quiet"])
    let complete_steps = [s_fmt, s_ssot, s_toestub, s_arch, s_clippy]
    let tier_complete = make_tier(complete_steps)

    // full tier adds nextest (no cov, no slow)
    print("\n[full tier (adds nextest, slow excluded)]")
    let s_nextest = run_and_record("nextest --workspace --run-ignored never", "cargo",
        ["nextest", "run", "--workspace", "--no-fail-fast", "--retries", "0", "--run-ignored", "never"])
    let full_steps = [s_fmt, s_ssot, s_toestub, s_arch, s_clippy, s_nextest]
    let tier_full = make_tier(full_steps)

    // full+cov tier replaces nextest with llvm-cov nextest
    print("\n[full+cov tier (llvm-cov, slow excluded)]")
    let s_cov = run_and_record("llvm-cov nextest --workspace --run-ignored never", "cargo",
        ["llvm-cov", "nextest", "--workspace", "--no-fail-fast", "--retries", "0",
         "--run-ignored", "never", "--no-report"])
    let full_cov_steps = [s_fmt, s_ssot, s_toestub, s_arch, s_clippy, s_cov]
    let tier_full_cov = make_tier(full_cov_steps)

    // ── doctests ──────────────────────────────────────────────────────────
    print("\n[doctests]")
    let s_doctests = run_and_record("cargo test --workspace --doc", "cargo", ["test", "--workspace", "--doc", "--quiet"])

    // ── assemble JSON output ──────────────────────────────────────────────
    let fast_ms = tier_fast.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
    let complete_ms = tier_complete.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
    let full_ms = tier_full.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
    let full_cov_ms = tier_full_cov.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)
    let doc_ms = s_doctests.get("wall_ms").and_then(fn(j: Json) to Option[int] { j.as_int() }).unwrap_or(0)

    // Get today's date for filename
    let date_res = process.run("powershell", ["-NoProfile", "-Command", "Get-Date -Format 'yyyy-MM-dd'"])
    let today = if date_res is null { "unknown" } else { date_res.unwrap().stdout.trim() }

    let out_path = "contracts/reports/test-baseline-" + today + ".json"

    // Build tier JSON strings
    let fast_json = json.parse("{\"wall_ms\":" + str(fast_ms) + ",\"steps\":[]}").unwrap()
    let _ = fast_json // suppress unused warning; real data is in tier_fast

    let tiers_str = "{\"fast\":" + str(fast_ms) + ",\"complete\":" + str(complete_ms) +
        ",\"full\":" + str(full_ms) + ",\"full_cov\":" + str(full_cov_ms) +
        ",\"doctests\":" + str(doc_ms) + "}"

    let report_str = "{\"schema_version\":1,\"generated_at\":\"" + today + "\",\"env\":{" +
        "\"cpu_logical_threads\":" + cpu_threads + "," +
        "\"ram_gb\":" + ram_gb + "," +
        "\"target_dir_gb\":" + target_gb + "," +
        "\"cargo_version\":\"" + cargo_ver + "\"," +
        "\"nextest_version\":\"" + nextest_ver + "\"," +
        "\"llvm_cov_version\":\"" + llvmcov_ver + "\"" +
        "},\"tiers\":" + tiers_str + "}"

    fs.write(out_path, report_str)
    print("\nReport written to: " + out_path)

    // ── Markdown summary ─────────────────────────────────────────────────
    print("\n=== Summary ===")
    print("Tier          | Wall-clock")
    print("------------- | ----------")
    print("fast          | " + str(fast_ms) + "ms")
    print("complete      | " + str(complete_ms) + "ms")
    print("full          | " + str(full_ms) + "ms")
    print("full+cov      | " + str(full_cov_ms) + "ms")
    print("doctests      | " + str(doc_ms) + "ms")
}
```

```rust
//! Persistent on-disk cache for arch-check results.
//!
//! Cache key: SHA-256 of Cargo.lock + layers.toml + where-things-live.md (concatenated).
//! Cache location: `<workspace_root>/target/vox-arch-check-cache/`.
//! Cache file: `<key>.json` — contains serialized `CachedData`.

use std::path::Path;
use anyhow::{Context, Result};
use sha2::{Digest, Sha256};
use serde::{Deserialize, Serialize};

/// The data we cache to avoid re-running expensive operations.
#[derive(Debug, Serialize, Deserialize)]
pub struct CachedData {
    /// SHA-256 of inputs (Cargo.lock + layers.toml + where-things-live.md).
    pub key: String,
    /// Result of the single `cargo metadata` call: package names + their
    /// manifest paths, dependency lists, and crate-type info. Stored as
    /// raw JSON string to avoid a hard dependency on cargo_metadata types here.
    pub metadata_json: String,
    /// Paths touched since last release date (Rule 8). `None` if git failed.
    pub git_touched_paths: Option<Vec<String>>,
    /// Last release version and date parsed from CHANGELOG.md.
    pub release_info: Option<(String, String)>,
}

/// Compute the cache key from the three input files.
pub fn compute_key(workspace_root: &Path) -> Result<String> {
    let mut hasher = Sha256::new();
    for rel in &[
        "Cargo.lock",
        "docs/src/architecture/layers.toml",
        "docs/src/architecture/where-things-live.md",
    ] {
        let path = workspace_root.join(rel);
        let contents = std::fs::read(&path)
            .with_context(|| format!("read {} for cache key", path.display()))?;
        hasher.update(&contents);
    }
    Ok(format!("{:x}", hasher.finalize()))
}

/// Return the cache file path for a given key.
fn cache_path(workspace_root: &Path, key: &str) -> std::path::PathBuf {
    workspace_root
        .join("target")
        .join("vox-arch-check-cache")
        .join(format!("{key}.json"))
}

/// Try to load a valid cache hit. Returns `None` if absent, stale, or corrupt.
pub fn load(workspace_root: &Path, key: &str) -> Option<CachedData> {
    let path = cache_path(workspace_root, key);
    let bytes = std::fs::read(&path).ok()?;
    let data: CachedData = serde_json::from_slice(&bytes).ok()?;
    if data.key != key {
        return None;
    }
    Some(data)
}

/// Persist cache data for a given key.
pub fn store(workspace_root: &Path, data: &CachedData) -> Result<()> {
    let path = cache_path(workspace_root, &data.key);
    if let Some(parent) = path.parent() {
        std::fs::create_dir_all(parent)?;
    }
    let bytes = serde_json::to_vec_pretty(data)?;
    std::fs::write(&path, bytes).with_context(|| path.display().to_string())?;
    Ok(())
}
```

```rust
/// Map changed file paths to the crate names that own them.
/// `pkg_manifest_paths`: list of (crate_name, relative manifest path).
/// Returns the set of crate names whose source tree was touched.
pub fn direct_crates_from_changes(
    changed_paths: &[String],
    pkg_manifest_paths: &[(String, String)],
) -> std::collections::HashSet<String> {
    let mut result = std::collections::HashSet::new();
    for (name, manifest) in pkg_manifest_paths {
        // Manifest is like "crates/vox-compiler/Cargo.toml"; crate root is its parent dir.
        let crate_prefix = std::path::Path::new(manifest)
            .parent()
            .map(|p| p.to_string_lossy().replace('\\', "/"))
            .unwrap_or_default();
        if crate_prefix.is_empty() {
            continue;
        }
        let prefix_slash = format!("{crate_prefix}/");
        for path in changed_paths {
            let norm = path.replace('\\', "/");
            if norm.starts_with(&prefix_slash) || norm == crate_prefix {
                result.insert(name.clone());
                break;
            }
        }
    }
    result
}
```

- [ ] **Step 3: Add `resolve_since_packages` function**

```rust
/// Resolve the set of packages to test when `--since <ref>` is specified.
///
/// Returns `None` if the number of directly-impacted packages exceeds
/// `SINCE_FALLBACK_THRESHOLD` (fall back to full workspace).
const SINCE_FALLBACK_THRESHOLD: usize = 20;

fn resolve_since_packages(root: &Path, git_ref: &str) -> Result<Option<Vec<String>>> {
    // 1. Get changed file paths since the ref
    let diff = Command::new("git")
        .args(["diff", "--name-only", &format!("{git_ref}...HEAD")])
        .current_dir(root)
        .output()
        .context("git diff --name-only")?;
    if !diff.status.success() {
        // If git fails (e.g. detached HEAD), fall back to full workspace
        return Ok(None);
    }
    let changed_paths: Vec<String> = String::from_utf8_lossy(&diff.stdout)
        .lines()
        .map(|l| l.trim().replace('\\', "/"))
        .filter(|l| !l.is_empty())
        .collect();
    if changed_paths.is_empty() {
        // No changes — still run at least a token set so pre-push doesn't silently skip tests
        return Ok(None);
    }

    // 2. Get package manifest paths via cargo metadata
    let meta = cargo_metadata::MetadataCommand::new()
        .no_deps()
        .exec()
        .context("cargo metadata for --since")?;

    let pkg_manifest_paths: Vec<(String, String)> = meta
        .workspace_packages()
        .iter()
        .map(|p| {
            let rel = p.manifest_path
                .strip_prefix(meta.workspace_root.as_std_path())
                .unwrap_or(p.manifest_path.as_std_path())
                .to_string_lossy()
                .replace('\\', "/");
            (p.name.clone(), rel)
        })
        .collect();

    // 3. Find directly-impacted crates
    let direct = direct_crates_from_changes(&changed_paths, &pkg_manifest_paths);
    if direct.is_empty() {
        // Changes are docs/scripts only; nothing to test
        return Ok(Some(Vec::new()));
    }

    // 4. Check fallback threshold
    if direct.len() > SINCE_FALLBACK_THRESHOLD {
        eprintln!(
            "pre-push --since: {} impacted crates exceeds threshold ({}); falling back to full workspace",
            direct.len(), SINCE_FALLBACK_THRESHOLD
        );
        return Ok(None);
    }

    Ok(Some(direct.into_iter().collect()))
}
```
